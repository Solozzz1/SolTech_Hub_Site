<script dangerouslySetInnerHTML={{
  __html: `
    const slideIndices = {
      ats: 0,
      solbot: 0
    };

    function showSlide(id, index) {
      const slideshow = document.querySelector(\`#\${id} .slides\`);
      const slides = slideshow.querySelectorAll('.slide');
      const totalSlides = slides.length;
      
      if (index >= totalSlides) {
        slideIndices[id] = 0;
      } else if (index < 0) {
        slideIndices[id] = totalSlides - 1;
      } else {
        slideIndices[id] = index;
      }
      
      slideshow.style.transform = \`translateX(-\${slideIndices[id] * 100}%)\`;
    }

    function nextSlide(id) {
      showSlide(id, slideIndices[id] + 1);
    }

    function prevSlide(id) {
      showSlide(id, slideIndices[id] - 1);
    }

    // Initialize slideshows
    document.addEventListener('DOMContentLoaded', () => {
      // Set initial positions
      showSlide('ats', 0);
      showSlide('solbot', 0);
      
      // Add auto-slide functionality
      setInterval(() => {
        nextSlide('ats');
      }, 5000);
      
      setInterval(() => {
        nextSlide('solbot');
      }, 5000);
    });
  `
